# Platform abstraction - static library
# Simple flat structure: all sources in module root, detail/ for PIMPL

add_library(sdk_platform STATIC
    window.cpp
    module.cpp
    input.cpp
)

target_include_directories(sdk_platform PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/..
)

target_compile_features(sdk_platform PUBLIC cxx_std_23)

set_target_properties(sdk_platform PROPERTIES
    POSITION_INDEPENDENT_CODE ON
    FOLDER "SDK"
)
